\begin{frame}{Tło: symulacja działa w klatkach}
  \begin{center}
  \begin{tikzpicture}[font=\small, scale=1.0]
    \fill[black!25] (3.0,-0.2) rectangle (3.35,2.4);
    \node[above] at (3.18,2.4) {ściana};
    \draw[thick, PwrRed, fill=PwrSand!60] (0.8,1.1) circle (0.35);
    \node[below=0.45cm] at (0.8,1.1) {klatka $t$};
    \draw[thick, PwrRed, fill=PwrSand!60] (5.6,1.1) circle (0.35);
    \node[below=0.45cm] at (5.6,1.1) {klatka $t+\Delta t$};
    \draw[-{Stealth}, very thick, PwrDark, dashed] (1.2,1.1) -- (5.2,1.1);
  \end{tikzpicture}
  \end{center}
  \begin{itemize}
    \item Kolizje sprawdzane tylko \textbf{w klatkach} (np. 60 razy na sekundę)
    \item Szybki obiekt może ,,przeskoczyć'' ścianę $\Rightarrow$ \textbf{tunelowanie}
    \item Przykład: pocisk 400 m/s przy 60 Hz $\to$ \textbf{$\approx 6{,}7$ m} między klatkami
    \item Rozwiązanie: \textbf{ciągła detekcja kolizji (CCD)} -- sprawdza całą drogę
  \end{itemize}
  \symbols{$t$ -- chwila czasu symulacji; $\Delta t$ -- krok czasowy symulacji (np.\ $1/60$\,s).}
  \src{Ericson, \textit{Real-Time Collision Detection}, 2004.}
  \nsi{Rysunek}
\end{frame}

\begin{frame}{Metoda 1: substepping}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize, scale=1.2]
        \fill[wall] (3.1,-0.1) rectangle (3.4,2.1);
        \foreach \x in {0.5,1.3,2.1} \draw[ghost] (\x,1.0) circle (0.3);
        \draw[ball] (2.82,1.0) circle (0.3);
        \node[PwrRed, font=\bfseries\small] at (2.82,1.6) {kontakt};
        \draw[decorate, decoration={brace, amplitude=3pt, mirror}] (0.5,0.55) -- (1.3,0.55)
          node[midway, below=3pt] {$\Delta t / n$};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}\itemsep0.4em
        \item[] \textbf{Idea:} krok $\Delta t$ dzielony na $n$ mniejszych kroków
        \item[] \textbf{Zaleta:} trywialna implementacja, nic nie trzeba zmieniać w detekcji
        \item[] \textbf{Wada:} koszt rośnie liniowo z $n$, brak gwarancji dla bardzo szybkich obiektów
      \end{itemize}
      \symbols{$\Delta t$ -- krok czasowy symulacji; $n$ -- liczba podkroków, na które dzielony jest krok.}
    \end{column}
  \end{columns}
  \src{Ericson 2004.}
  \nsi{Rysunek}
\end{frame}

      \end{itemize}
      \symbols{$d_1, d_2$ -- kolejne bezpieczne przesunięcia, wyznaczane z aktualnej odległości od przeszkody.}
    \end{column}
  \end{columns}

\begin{frame}{Metoda 4: speculative contacts}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \centering
      \begin{tikzpicture}[font=\scriptsize, scale=1.2]
        \fill[wall] (3.1,-0.1) rectangle (3.4,2.1);
        \draw[ball] (1.2,1.0) circle (0.3);
        \draw[ghost] (4.5,1.0) circle (0.3);
        \draw[mv, dashed] (1.55,1.0) -- (4.15,1.0);
        \draw[PwrRed, very thick] (1.5,1.0) -- (3.1,1.0);
        \node[PwrRed, font=\small\bfseries, above] at (2.3,1.15) {kontakt ,,na zapas''};
        \node[below] at (2.3,0.75) {dystans $\le v\,\Delta t$};
      \end{tikzpicture}
      \symbols{$v$ -- prędkość obiektu;\\ $v\,\Delta t$ -- droga pokonywana w jednym kroku.}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}\itemsep0.4em
        \item[] \textbf{Idea:} kontakt tworzony z wyprzedzeniem, gdy przeszkoda jest w zasięgu ruchu; solver nie pozwala jej przekroczyć
        \item[] \textbf{Zaleta:} tanie, bez dodatkowych iteracji
        \item[] \textbf{Wada:} ,,ghost collisions'' -- zatrzymanie przed przeszkodą, której by nie dotknął
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  {\color{PwrRed}\bfseries Każda metoda to inny kompromis:} niezawodność -- koszt -- artefakty
  \src{Catto, GDC 2013; dokumentacja Box2D, Jolt Physics.}
  \nsi{Rysunek}
\end{frame}

\begin{frame}{Pytania badawcze}
  \Large\setlength{\leftmargini}{2.6em}
  \begin{enumerate}\itemsep0.7em
    \item[\textbf{PB1}] Która metoda ma najmniejszy koszt obliczeniowy potrzebny do uzyskania 100\% niezawodności zależnie od prędkości~$v$ i liczby obiektów N?
    \item[\textbf{PB2}] Jak liczba fałszywych kontaktów wprowadzanych przez poszczególne metody zależy od prędkości~$v$ i prędkości kątowej~$\omega$?
  \end{enumerate}
\end{frame}
 
% 9
\begin{frame}{Hipotezy}
  \setlength{\leftmargini}{2.2em}
  \begin{enumerate}\itemsep0.7em
    \item[\textbf{H1}] (PB1) Dla $\omega = 0$ i $v\,\Delta t > d + s$ \textbf{sweep}
      wymaga mniejszego kosztu niż \textbf{substepping} przy każdym badanym $N$.
    \item[\textbf{H2}] (PB1) \textbf{Speculative contacts} nie osiąga niezawodności
      100\% w co najmniej jednym badanym scenariuszu.
    \item[\textbf{H3}] (PB2) Liczba fałszywych kontaktów rośnie z $v$ i $\omega$
      tylko w \textbf{speculative contacts}; w pozostałych metodach wynosi 0.
  \end{enumerate}
  \vfill
  \symbols{$\Delta t$ -- krok czasowy; $d$ [m] -- grubość przeszkody; $s$ [m] -- rozmiar obiektu w~kierunku ruchu;\\
    $v\,\Delta t > d + s$ -- obiekt w~jednym kroku pokonuje drogę większą niż przeszkoda wraz z~obiektem (możliwe tunelowanie).}
  \vspace{0.2cm}
\end{frame}
